# ===============================
# 1. 라이브러리 불러오기
# ===============================
import pandas as pd
import matplotlib.pyplot as plt

plt.rcParams['axes.unicode_minus'] = False

# ===============================
# 2. CSV 파일 불러오기
# ===============================
file_path = '서울교통공사_지하철혼잡도정보.csv'
df = pd.read_csv(file_path, encoding='cp949')

# ===============================
# 3. 데이터 필터링
# ===============================
filtered_df = df[
    (df['요일구분'] == '평일') &
    (df['호선'] == '7호선')
]

up_df = filtered_df[filtered_df['상하구분'] == '상선']
down_df = filtered_df[filtered_df['상하구분'] == '하선']

# ===============================
# 4. 시간 컬럼 설정 (원본)
# ===============================
time_columns = [
    '5시30분','6시00분','6시30분','7시00분','7시30분','8시00분','8시30분',
    '9시00분','9시30분','10시00분','10시30분','11시00분','11시30분',
    '12시00분','12시30분','13시00분','13시30분','14시00분','14시30분',
    '15시00분','15시30분','16시00분','16시30분','17시00분','17시30분',
    '18시00분','18시30분','19시00분','19시30분','20시00분','20시30분',
    '21시00분','21시30분','22시00분','22시30분','23시00분','23시30분',
    '00시00분','00시30분'
]

# ===============================
# 5. 시간 문자열을 HH:MM 형식으로 변환하는 함수
# ===============================
def convert_time(korean_time):
    # '5시30분' → ['5', '30']
    hour = korean_time.split('시')[0]
    minute = korean_time.split('시')[1].replace('분', '')
    
    # 두 자리로 맞추기 (05:30 형태)
    return f"{int(hour):02d}:{int(minute):02d}"

# 변환된 시간 리스트 생성
time_labels = [convert_time(t) for t in time_columns]

# ===============================
# 6. 평균 혼잡도 계산
# ===============================
up_mean = up_df[time_columns].mean()
down_mean = down_df[time_columns].mean()

# ===============================
# 7. 그래프 그리기
# ===============================
plt.figure(figsize=(14, 6))

plt.plot(time_labels, up_mean, marker='o', label='Up Line')
plt.plot(time_labels, down_mean, marker='s', label='Down Line')

plt.title('Line 7 Weekday Congestion by Time (Up vs Down)')
plt.xlabel('Time (HH:MM)')
plt.ylabel('Congestion Level')

plt.xticks(rotation=45)
plt.legend()
plt.grid()

plt.tight_layout()
plt.show()
